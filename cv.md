  const result = {};
  let repeats = 0;
  text
  .split('')
  .map(a => a.toUpperCase())
  .filter(a => !a.match(/\W|_/g))
  .forEach(item => typeof result[item] === 'number' ? result[item]++ : result[item] = 0);

  }
